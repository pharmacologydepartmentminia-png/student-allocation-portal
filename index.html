function doGet(e) { var ss = SpreadsheetApp.getActiveSpreadsheet(); // 1. Fetch Students data (assuming main sheet is first tab) var studentSheet = ss.getSheets()[0]; var studentRows = studentSheet.getDataRange().getValues(); var students = []; // Skip header row for (var i = 1; i < studentRows.length; i++) { if (studentRows[i][1]) { // Ensure student name exists students.push({ id: studentRows[i][0], // Column A: ID name: studentRows[i][1], // Column B: Name wave: studentRows[i][3] // Column D: Wave }); } } // 2. Fetch Departments data from 'departments' sheet var deptSheet = ss.getSheetByName("departments"); var depts = []; if (deptSheet) { var deptRows = deptSheet.getDataRange().getValues(); for (var j = 1; j < deptRows.length; j++) { if (deptRows[j][0]) { depts.push(deptRows[j][0]); // Column A: Department Name } } } var result = { students: students, departments: depts }; return ContentService .createTextOutput(JSON.stringify(result)) .mimeType(ContentService.MimeType.JSON); } function doPost(e) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName("Submissions") || ss.getSheets()[0]; var data = JSON.parse(e.postData.contents); sheet.appendRow([ new Date(), data.wave, data.id, data.name, data.preferences.join(', ') ]); return ContentService .createTextOutput(JSON.stringify({ "result": "success" })) .mimeType(ContentService.MimeType.JSON); }
